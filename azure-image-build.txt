1.Create Resource Groups
---------------------------
az group create -n packer-rg -l switzerlandnorth
az group create -n vsphere-rg -l switzerlandnorth

az group show -n vsphere-rg

delete:
----------
az group delete --name vsphere-rg --yes --no-wait
az group delete --name packer-rg --yes --no-wait


2.Create Azure Compute Gallery

az sig create \
  --resource-group vsphere-rg \
  --gallery-name vsphere_gallery

az sig show -g vsphere-rg -n vsphere_gallery

 3. Create Image Definition
az sig image-definition create \
  --resource-group vsphere-rg  \
  --gallery-name vsphere_gallery \
  --gallery-image-definition vsphere-ubuntu-image \
  --publisher vspheretech \
  --offer myOffer \
  --sku ubuntu-22.04 \
  --os-type Linux
  
  
az sig image-definition show \
  -g vsphere-rg \
  --gallery-name vsphere_gallery \
  --gallery-image-definition vsphere-ubuntu-image

4. Create Service Principal
